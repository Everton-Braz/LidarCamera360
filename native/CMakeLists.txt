cmake_minimum_required(VERSION 3.24)
project(RavenNative VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Eigen3 CONFIG REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenCV CONFIG REQUIRED)
find_package(yaml-cpp CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
set(SOPHUS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps-src/Sophus" CACHE PATH "Legacy Sophus source (a621ff)")
set(VIKIT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps-src/rpg_vikit" CACHE PATH "Vikit source")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
  --upstream "${CMAKE_CURRENT_SOURCE_DIR}/../FAST-LIVO2" --output "${CMAKE_CURRENT_BINARY_DIR}/port"
  --sophus "${SOPHUS_SOURCE}" --vikit "${VIKIT_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
set(PORT "${CMAKE_CURRENT_BINARY_DIR}/port")
add_library(legacy_sophus STATIC "${PORT}/sophus/sophus/se3.cpp" "${PORT}/sophus/sophus/so3.cpp")
target_include_directories(legacy_sophus PUBLIC "${PORT}/sophus")
target_link_libraries(legacy_sophus PUBLIC Eigen3::Eigen)
add_library(vikit_native STATIC
  "${PORT}/vikit/src/pinhole_camera.cpp"
  "${PORT}/vikit/src/equidistant_camera.cpp" "${PORT}/vikit/src/math_utils.cpp"
  "${PORT}/vikit/src/vision.cpp" "${PORT}/vikit/src/robust_cost.cpp")
target_include_directories(vikit_native PUBLIC "${PORT}/vikit/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vikit_native PUBLIC legacy_sophus ${OpenCV_LIBS})
add_executable(fastlivo2 main.cpp "${PORT}/src/LIVMapper.cpp" "${PORT}/src/IMU_Processing.cpp"
  "${PORT}/src/voxel_map.cpp" "${PORT}/src/vio.cpp" "${PORT}/src/frame.cpp" "${PORT}/src/visual_point.cpp")
target_include_directories(fastlivo2 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PORT}/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(fastlivo2 PRIVATE vikit_native ${PCL_LIBRARIES} yaml-cpp::yaml-cpp OpenMP::OpenMP_CXX)
target_compile_definitions(fastlivo2 PRIVATE FASTLIVO_STANDALONE MP_EN MP_PROC_NUM=4 ROOT_DIR=native::output_root.c_str\(\))
foreach(t legacy_sophus vikit_native fastlivo2)
  target_compile_definitions(${t} PRIVATE NOMINMAX _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    target_compile_options(${t} PRIVATE /O2 /bigobj /EHsc /utf-8 /FIcassert)
  else()
    target_compile_options(${t} PRIVATE -O3)
  endif()
endforeach()
install(TARGETS fastlivo2 RUNTIME DESTINATION bin)
